-- 1) SESSIONS WHATSAPP : une ligne par numéro client, avec l'état de conversation.
create table if not exists public.sessions_whatsapp (
  id uuid primary key default gen_random_uuid(),
  wa_id text not null unique,
  phone_number text,
  display_name text,
  language text,
  session_state text not null default 'NEW',
  conversation_context jsonb not null default '[]'::jsonb,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);
alter table public.sessions_whatsapp enable row level security;

-- 2) RENDEZ-VOUS : créés par l'agent, confirmés par le paiement des 50 €.
create table if not exists public.rendez_vous (
  id uuid primary key default gen_random_uuid(),
  wa_id text not null,
  customer_name text,
  phone text,
  device_model text,
  repair_type text,
  requested_date text,
  requested_time text,
  estimated_price text,
  deposit_amount numeric not null default 50,
  payment_status text not null default 'PENDING' check (payment_status in ('PENDING','PAID','FAILED','CANCELLED')),
  appointment_status text not null default 'AWAITING_PAYMENT' check (appointment_status in ('AWAITING_PAYMENT','CONFIRMED','MODIFIED','CANCELLED','DONE')),
  notes text,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);
alter table public.rendez_vous enable row level security;

-- 3) CATALOGUE RÉPARATIONS : les prix par modèle. AUCUNE ligne inventée :
--    seules les règles sûres du brief sont insérées ci-dessous ; la grille de
--    Franck s'ajoutera ici sans changer le moteur.
create table if not exists public.catalogue_reparations (
  id uuid primary key default gen_random_uuid(),
  brand text,
  model text,               -- 'ANY' = valable quel que soit le modèle
  repair_type text not null,
  part_type text,           -- 'fournie_client' | 'origine' | 'compatible' | null
  price numeric,            -- prix pièce si connu
  labor_price numeric,      -- main-d'œuvre si connue
  shipping_cost numeric,
  margin numeric,
  total_customer_price numeric,  -- LE prix annonçable au client ; null = inconnu
  price_is_minimum boolean not null default false, -- true = « à partir de » (jamais un prix ferme)
  delivery_delay text,
  warranty text,
  source text not null,
  active boolean not null default true,
  updated_at timestamptz not null default now()
);
alter table public.catalogue_reparations enable row level security;

-- 4) ESCALADES : tout ce qui doit remonter à Franck.
create table if not exists public.escalades (
  id uuid primary key default gen_random_uuid(),
  wa_id text,
  customer_name text,
  reason text not null,
  summary_fr text,
  priority text not null default 'NORMAL' check (priority in ('LOW','NORMAL','HIGH')),
  status text not null default 'OPEN' check (status in ('OPEN','RESOLVED')),
  created_at timestamptz not null default now(),
  resolved_at timestamptz
);
alter table public.escalades enable row level security;

create index if not exists idx_sessions_wa on public.sessions_whatsapp (wa_id);
create index if not exists idx_rdv_wa on public.rendez_vous (wa_id);
create index if not exists idx_catalogue_lookup on public.catalogue_reparations (repair_type, model, active);
